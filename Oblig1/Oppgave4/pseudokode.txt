og ikke det minste.

input: et heap med heltall
output: printe elementene i en sortert rekkefølge     slik at treet vil bli balanseert

Procedure OmOrganiserHeap(H){
    if H = null then
        return
    midt <- (size-1) /2
    teller <- 0
    for tall: H do
        if teller <= midt then
            lavHeap.offer(tall)
        else
            hoyHeap.offer(tall)
        teller ++
    print(lavHeap.poll)
    OmOrganiserHeap(lavHeap)
    OmOrganiserHeap(hoyHeap)
}
