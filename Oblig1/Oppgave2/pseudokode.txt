procedure push_front
    ny <- new Node(x)

    if start = null then
        start <- ny
        middle <- ny
        end <- ny

    ny.next = start
    start.next = ny

    size ++

procedure push_back
    ny <- new Node(x)

    if start = null then
        start <- ny
        middle <- ny
        end <- ny

    end.next = ny
    end = ny

    size ++

procedure push_middle
    kopi <- middle.next

    if start = null then
        start <- ny
        middle <- ny
        end <- ny
        
    middle.next = ny
    ny.next = kopi

    size ++

procedure get(i)
    if i = 0 then
        return start
    tmp <- start
    a <- 0
    while a < i do
        tmp = tmp.next
        a++
    return tmp
